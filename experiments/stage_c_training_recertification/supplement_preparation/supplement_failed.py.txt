ORIGINAL=c.ROOT/'ml/models/stage_c_training_recertification/20260915T143739Z'
ORDER=['CPU_N_target','CPU_N_gradient','CUDA_N_input','CUDA_N_target','CUDA_N_gradient','serialization']

def inventory(root):
    return {str(p.relative_to(root)):c.sha(p) for p in sorted(root.rglob('*')) if p.is_file()}

def serialization(root,out,report,fixture,args,p,b,r,cpu,exact,close,count,torch,pf):
    """Original trained BESTs and saved interrupted LASTs, on verified copies only."""
    def deny(*a,**kw):
        raise AssertionError('serialization forbids fit/backward/base AdamW')
    def deny_adam(*a,**kw):
        ledger=json.loads((root/'ledger.json').read_text());ledger['attempts']+=1;c.save(root/'ledger.json',ledger)
        raise AssertionError('serialization containment: base AdamW forbidden')
    from unittest.mock import patch
    import lightning.pytorch as pl
    old=c.serialization_original
    with patch.object(torch.optim.AdamW,'step',deny_adam),patch.object(torch.Tensor,'backward',deny),patch.object(pl.Trainer,'fit',deny):
        # Registry.verified checks ownership, hash, schema, normalizer, Q and full state.
        last_evidence={}
        for name in ('CPU_P','CUDA_P'):
            saved=json.loads((root/name/'result.json').read_text())
            reg=r.Registry(root/'boundary_owned')
            path,payload,record=reg.verified(saved['run_id'],'last',saved['contract'])
            assert record==saved['registry']['last']
            assert record['global_step']==4 and record['boundary_complete'] and not record['terminated']
            assert payload['global_step']==4
            last_evidence[name]=dict(file=str(path.relative_to(root)),sha256=c.sha(path),role='last',step=4)
        report['last_verification']=last_evidence
        code=old(root,out,report,fixture,args,p,b,r,cpu,exact,close,count,torch,pf)
        assert json.loads((root/'ledger.json').read_text())['attempts']==0
        return code
